SYNTAX:
--NUMBER DATAYPE---------
NUMBER(PRECISION,[SCALE])


--HAVING CLAUSE----------
SELECT GROUP FUNCTION/GROUP BY EXPRESSION
FROM TABLE_NAME
[WHERE <FILTER CONDITION>]
GROUP BY COL_NAME/EXPRESSION
HAVING <GROUP FILTER CONDITION>;


----ORDER BY WITHOUT GROUP BY CLAUSE--------------
SELECT COL_NAME/EXPRESSION
FROM TABLE_NAME
[WHERE <FILTER CONDITION>]
ORDER BY COL_NAME/EXPRESSION [ASC]/DESC;


---NOT IN OPERATOR----------
COL_NAME/EXP  NOT IN (VAL1,VAL2....);


---NOT LIKE OPERATOR---------
COL_NAME/EXP NOT LIKE '_%CHARACTER%_';

---IS OPERATOR------------
COL_NAME/EXP IS NULL;


---INNER JOIN----------
SELECT COL_NAME/EXP
FROM TABLE_NAME1,TABLE_NAME2
WHERE <JOIN CONDITION>


--NATURAL JOIN------------
SELECT COL_NAME/EXPRESSION
FROM TABLE_NAME1 NATURAL JOIN TABLE_NAME2


--- RIGHT OUTER JOIN------------
SELECT COL_NAME/EXPRESSION
FROM TABLE_NAME1 , TABLE_NAME2
WHERE TABLE_NAME1.COL_NAME(+)=TABLE_NAME2.COL_NAME;
