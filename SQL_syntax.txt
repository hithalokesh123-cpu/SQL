----INNER JOIN-----------

FROM TABLE_NAME1,TABLE_NAME2

------NATURAL JOIN-------
SELECT COL_NAME/EXPRESSION
FROM TABLE_NAME1 NATURAL JOIN TABLE_NAME2

------LEFT OUTER JOIN---------
SELECT COL_NAME/EXP
FROM TABLE_NAME1,TABLE_NAME2
WHERE TABLE_NAME1.COL_NAME=TABLE_NAME2.COL_NAME(+);

----RIGHT OUTER JOIN-----
SELECT COL_NAME/EXP
FROM TABLE_NAME1,TABLE_NAME2
WHERE TABLE_NAME1.COL_NAME(+)=TABLE_NAME2.COL_NAME;

---FULL OUTER JOIN----------
SELECT COL_NAME/EXPRESSION
FROM TABLE_NAME1 FULL [OUTER] JOIN TABLE_NAME2
ON <JOIN CONDITION>

-------SELF JOIN-------

FROM TABLE_NAME1 JOIN TABLE_NAME2
ON <JOIN CONDITION>


---------COR-RELATED SUBQUERY--------
